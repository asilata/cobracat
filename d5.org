  def find_top(obj, ssc):
    """
    Compute the top HN piece of the given object with respect to the stability condition being considered.

    INPUT:

    - `obj` : a ProjectiveComplex
    - `stab`: a sequence of objects ordered by increasing phase.

    OUTPUT:

    A tuple (x, h) where x is a stable object of the highest phase that has a non-trivial morphism to obj.
    The morphism h is one such morphism.
    """

    stab = list(reversed(flatten([x[1] for x in ssc.phase_order()])))
    
    top_shift = -Infinity
    top_object = None
    bottom_shift = +Infinity
    
    rawHoms = {}
    for i in range(0, len(stab)):
      stable = stab[i]
      rawHoms[i] = stable.hom(obj, unreduced=True, collapsed_grading=True)
      if -rawHoms[i].min_index > top_shift:
          top_shift = -rawHoms[i].min_index
          top_object = i
      if -rawHoms[i].max_index < bottom_shift:
        bottom_shift = -rawHoms[i].max_index

    
    while True:
      betti = len(rawHoms[top_object][-top_shift]) - matrix(rawHoms[top_object].maps.get(-top_shift,{})).rank() - matrix(rawHoms[top_object].maps.get(-top_shift-1,{})).rank()

      if betti > 0:
        break
      
      if top_shift < bottom_shift:
        raise ValueError(f"Could not find top of {obj}")

      if top_object < len(stab)-1:
        top_object = top_object + 1
      else:
        top_object = 0
        top_shift = top_shift - 1


    (h, hom_data)= stab[top_object].hom(obj, with_homs=True)
    for i in range(h.min_index, h.max_index+1):
      for j in range(0, len(h[i])):
        if h[i][j].graded_degree - i == top_shift:
          highest_object_found = stab[top_object].graded_shift(h[i][j].graded_degree).homological_shift(-i)
          highest_hom_found = {x+i : hom_data[i][j][x] for x in hom_data[i][j]}
                    
          return (highest_object_found, highest_hom_found)

    raise ValueError("Could not find top!")
